Overview

Three boxes' worth of electronics in the v4 case, three independent chains. The Pi and the beacon share two wires (W9 signal, W10 ground) and nothing else; the Pi only listens. The SMS backup shares nothing.

GY-63 (MS5611): only VCC, GND, SDA, SCL. PS, CSB and SDO stay unconnected. That the board pulls PS high (I2C) and CSB low (address 0x77) is an assumption; the check is sudo i2cdetect -y 1 showing 77 (or 76). Nothing at all means PS is not pulled up: solder PS to VCC on the GY-63 itself.

Heltec to Pi tap line

One line a second on W9, 38400 8N1, Heltec UART2 TX only. Checksum: XOR of every character between $ and *. Field meanings: pi/core/.../TapLine.kt.

$HAB1,<seq>,<up_ms>,<boots>,<rst>,<fix_age_ms>,<utc>,<tx_dbm>,<nav>,<frame_hex>*HH

Example (the golden line the C++ and Kotlin tests share):

$$HAB1,513,123456,7,1,850,20261031063015,22,3,542AF9C58D414BAF9742009CDE4500341E471EFB0042DD450B0F930F*14"
const val GOLDEN_2 = "\$HAB1,0,900,1,9,-1,-,-9,0,542AF9C58D414BAF9742009CDE4500341E471EFB0042DD450B0F930F*3C"

class Ms5611Test {
    @Test fun datasheetExample() {
        val (t, p) = Ms5611.compensate(C, 9085466, 8569150)
        assertEquals(2007, t)          // 20.07 C
        assertEquals(100009, p)        // 1000.09 mbar
    }

    /** Straight transcription of the datasheet formulas in Double, divisions not shifts. */
    private fun reference(c: IntArray, d1: Long, d2: Long): Pair<Double, Double> {
        val dT = d2 - c[5] * 256.0
        var temp = 2000 + dT * c[6] / 2.0.pow(23)
        var off = c[2] * 2.0.pow(16) + c[4] * dT / 2.0.pow(7)
        var sens = c[1] * 2.0.pow(15) + c[3] * dT / 2.0.pow(8)
        if (temp < 2000) {
            val t2 = dT * dT / 2.0.pow(31)
            var off2 = 5 * (temp - 2000).pow(2) / 2; var sens2 = 5 * (temp - 2000).pow(2) / 4
            if (temp < -1500) { off2 += 7 * (temp + 1500).pow(2); sens2 += 11 * (temp + 1500).pow(2) / 2 }
            temp -= t2; off -= off2; sens -= sens2
        }
        return temp to (d1 * sens / 2.0.pow(21) - off) / 2.0.pow(15)
    }

    @Test fun coldCaseMatchesReferenceWithoutOverflow() {
        // D2 chosen for about -50 C, D1 for a low pressure: both second-order branches run.
        for ((d1, d2) in listOf(5_000_000L to 6_492_706L, 6_500_000L to 7_600_000L, 9_085_466L to 8_569_150L)) {
            val (t, p) = Ms5611.compensate(C, d1, d2)
            val (rt, rp) = reference(C, d1, d2)
            assertTrue(abs(t - rt) <= 2.0, "temp $t vs reference $rt")
            assertTrue(abs(p - rp) <= 2.0, "pressure $p vs reference $rp")
        }
        val (t, _) = Ms5611.compensate(C, 5_000_000L, 6_492_706L)
        assertTrue(t < -4000, "expected a cold reading, got $t")
    }

    private fun withCrc(w: IntArray): IntArray { val x = w.copyOf(); x[7] = (x[7] and 0xFFF0) or Ms5611.crc4(x); return x }

    @Test fun crcAcceptsGoodPromAndRejectsOneFlippedBit() {
        val good = withCrc(intArrayOf(0x0002, 40127, 36924, 23317, 23282, 33464, 28312, 0x1230))
        assertTrue(Ms5611.crcOk(good))
        for (word in 0..6) for (bit in 0..15) {
            val bad = good.copyOf(); bad[word] = bad[word] xor (1 shl bit)
            assertFalse(Ms5611.crcOk(bad), "flip word $word bit $bit not caught")
        }
    }

    /** A fake bus that behaves like an MS5611 at one address. */
    private class FakeMs(val addr: Int, val prom: IntArray, val d1: Long, val d2: Long) : I2cBus {
        var pending = 0L; var zeroNext = false
        override fun write(address: Int, bytes: ByteArray) {
            if (address != addr) throw RuntimeException("nack")
            when (bytes[0].toInt() and 0xFF) { 0x48 -> pending = d1; 0x58 -> pending = d2 }
        }
        override fun readRegister(address: Int, reg: Int, count: Int): ByteArray {
            if (address != addr) throw RuntimeException("nack")
            if (reg in 0xA0..0xAE) { val w = prom[(reg - 0xA0) / 2]; return byteArrayOf((w shr 8).toByte(), w.toByte()) }
            if (reg == 0x00) { val v = if (zeroNext) 0L else pending; zeroNext = false; pending = 0; return byteArrayOf((v shr 16).toByte(), (v shr 8).toByte(), v.toByte()) }
            return ByteArray(count)
        }
        override fun scan() = listOf(addr)
        override fun close() {}
    }

    @Test fun driverFindsTheSensorAndReadsTheExample() {
        val prom = withCrc(intArrayOf(0x0002, 40127, 36924, 23317, 23282, 33464, 28312, 0x1230))
        val bus = FakeMs(0x76, prom, 9085466, 8569150)
        val m = Ms5611.probe(bus, {})
        assertNotNull(m); assertEquals(0x76, m.address); assertTrue(m.promCrcOk)
        val r = m.read()
        assertEquals(100009.0, r.pressurePa); assertEquals(20.07, r.temperatureC)
        bus.zeroNext = true
        assertFailsWith<IllegalStateException> { m.read() }   // an unfinished conversion is an error, not a reading
    }
}

class TapLineTest {
    private val frame = TelemetryFrame(42, 17.721666f, 75.84237f, 7123.5f, 40500f, -1250, 7080.25f, 11, 0x0F, 3987)

    @Test fun encodesExactlyLikeThePythonReference() {
        val t = TapLine(513, 123456, 7, 1, 850, "20261031063015", 22, 3, frame.encode())
        assertEquals(GOLDEN_1, TapLine.encode(t))
        val t2 = TapLine(0, 900, 1, 9, -1, null, -9, 0, frame.encode())
        assertEquals(GOLDEN_2, TapLine.encode(t2))
    }

    @Test fun parsesAndDecodesTheFrame() {
        val p = TapLine.parse(GOLDEN_1 + "\r\n")
        assertIs<TapLine.Parse.Ok>(p)
        assertEquals(frame, p.line.frame)
        assertEquals("20261031063015", p.line.utc)
        assertEquals(850, p.line.fixAgeMs)
        val p2 = TapLine.parse(GOLDEN_2)
        assertIs<TapLine.Parse.Ok>(p2); assertNull(p2.line.utc); assertEquals(-1, p2.line.fixAgeMs)
    }

    @Test fun acceptsTheFirstLineAfterAHeltecPowerUp() {
        // as received on the Pi 2026-10-27: NUL bytes before the '$' on the first line of a boot
        val p = TapLine.parse("\u0000\u0000\u0000" + GOLDEN_1)
        assertIs<TapLine.Parse.Ok>(p); assertEquals(513, p.line.seq)
        assertIs<TapLine.Parse.Bad>(TapLine.parse("\u0000\u0000" + GOLDEN_1.replace("*14", "*15")), "checksum still guards it")
    }

    @Test fun rejectsDamage() {
        // one character changed anywhere in the body must fail the checksum
        for (i in 1 until GOLDEN_1.indexOf('*')) {
            val c = GOLDEN_1[i]; val r = if (c == '1') '2' else '1'
            if (c == r) continue
            val bad = GOLDEN_1.substring(0, i) + r + GOLDEN_1.substring(i + 1)
            assertIs<TapLine.Parse.Bad>(TapLine.parse(bad), "change at $i accepted: $bad")
        }
        assertIs<TapLine.Parse.Bad>(TapLine.parse(GOLDEN_1.substring(0, 40)))          // torn line
        assertIs<TapLine.Parse.Bad>(TapLine.parse("\$GNGGA,123519,4807.038,N*47"))      // stray NMEA
        assertIs<TapLine.Parse.Bad>(TapLine.parse(""))
    }
}

class AltitudeGateTest {
    @Test fun oneSpikeOnThePadDoesNotStartIt() {
        val g = AltitudeGate()
        repeat(9) { assertFalse(g.update(7500.0, 7500.0)) }
        assertFalse(g.update(480.0, 7500.0))                 // run broken
        repeat(9) { assertFalse(g.update(7500.0, 7500.0)) }
        assertFalse(g.latched)
        assertTrue(g.update(7500.0, 7500.0))                 // 10th consecutive
        assertTrue(g.latched)
        assertFalse(g.update(100.0, 100.0)); assertTrue(g.latched)   // stays on
    }

    @Test fun bothMustAgreeUnlessOneIsWellAbove() {
        val g = AltitudeGate()
        repeat(30) { g.update(7400.0, 6900.0) }
        assertFalse(g.latched, "baro below 7 km holds it while GPS is under the 9 km override")
        repeat(10) { g.update(9100.0, 6900.0) }
        assertTrue(g.latched, "GPS above 9 km overrides a baro that reads low")
    }

    @Test fun singleSourceDecidesAlone() {
        val a = AltitudeGate(); repeat(10) { a.update(null, 7001.0) }; assertTrue(a.latched)
        val b = AltitudeGate(); repeat(10) { b.update(7001.0, null) }; assertTrue(b.latched)
        val c = AltitudeGate(); repeat(100) { c.update(null, null) }; assertFalse(c.latched)
        val d = AltitudeGate(); repeat(10) { d.update(Double.NaN, 6000.0) }; assertFalse(d.latched)
    }

    // ---- GPS fallback: 20 min after launch (or after GPS was lost), never on the pad ----
    private val MIN = 60_000L

    @Test fun padWithoutGpsNeverStarts() {
        val g = AltitudeGate()
        for (s in 0..7200) assertFalse(g.update(null, 509.0, s * 1000L), "started on the pad at $s s")
        assertNull(g.launchedAtMs)
    }

    @Test fun noGpsAndAFrozenBaroStartsTwentyMinutesAfterLaunch() {
        // baro freezes at 2,000 m (a failing sensor), so it never shows 7 km; GPS absent all flight.
        // (A baro that merely reads low still reaches its own "7 km" before the timer, and the 7 km rule wins.)
        val g = AltitudeGate()
        var latchedAt = -1L
        for (s in 0..4000) {
            val trueAlt = 509.0 + 5.0 * s
            if (g.update(null, minOf(trueAlt, 2000.0), s * 1000L)) { latchedAt = s * 1000L; break }
        }
        assertEquals(148_000L, g.launchedAtMs, "launch = first 10 s at or above 1,200 m: 139..148 s")
        val launch = g.launchedAtMs!!
        assertEquals(launch + 20 * MIN, latchedAt, "latched at the wrong time")
        assertTrue(g.lastReason.contains("GPS altitude unavailable"), g.lastReason)
    }

    @Test fun gpsLostMidAscentWithNoBaroStartsTwentyMinutesAfterTheLoss() {
        val g = AltitudeGate()
        var latchedAt = -1L
        for (s in 0..6000) {
            val alt = 509.0 + 5.0 * s
            val gps = if (alt < 3000.0) alt else null          // GPS gives up at 3 km, baro dead
            if (g.update(gps, null, s * 1000L)) { latchedAt = s * 1000L; break }
        }
        assertEquals(g.gpsLastSeenMs!! + 20 * MIN, latchedAt)
        assertTrue(g.lastReason.contains("since GPS was lost"), g.lastReason)
    }

    @Test fun gpsPresentButLowDoesNotUseTheFallback() {
        val g = AltitudeGate()
        for (s in 0..7200) g.update(5000.0, 5000.0, s * 1000L)   // stuck below 7 km for 2 h, GPS present
        assertFalse(g.latched)
    }

    @Test fun sevenKilometresStillWinsWhenTheBaroIsRight() {
        // GPS absent, baro correct: 7 km at (7000-509)/5 = 1298 s, before launch+20 min (about 148 + 1200 s)
        val g = AltitudeGate()
        var latchedAt = -1L
        for (s in 0..4000) if (g.update(null, 509.0 + 5.0 * s, s * 1000L)) { latchedAt = s * 1000L; break }
        assertTrue(latchedAt in 1298_000L..1310_000L, "latched at $latchedAt")
        assertTrue(g.lastReason.contains("baro only"), g.lastReason)
    }

    @Test fun restoredLatchStaysOn() {
        val g = AltitudeGate(latched = true)
        assertFalse(g.update(400.0, 400.0)); assertTrue(g.latched)
    }
}
